<script>
"use strict";
__TOOL_CONFIG__

/* =====================================================================
 * 通用引擎
 * ===================================================================== */

const LS_DRAFT = TOOL.ls + "_draft";
const LS_REPORTS = TOOL.ls + "_reports";
const STEPS = TOOL.steps;

let view = "home";
let cur = 0;
let values = {};
let reports = loadReports();
let viewingReport = null;
let resetScrollNext = true;
let reviewEdit = -1;
let reviewIdx = null;

function loadReports() {
  try { return JSON.parse(localStorage.getItem(LS_REPORTS)) || []; } catch (e) { return []; }
}
function saveDraft() {
  try { localStorage.setItem(LS_DRAFT, JSON.stringify(values)); } catch (e) {}
}
function loadDraft() {
  try { return JSON.parse(localStorage.getItem(LS_DRAFT)) || null; } catch (e) { return null; }
}
function persistReports() {
  try { localStorage.setItem(LS_REPORTS, JSON.stringify(reports)); } catch (e) {}
}

function esc(s) {
  return String(s == null ? "" : s)
    .replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;");
}

function optionsOf(vals) {
  return (Array.isArray(vals.options) ? vals.options : []).filter(o => (o.t || "").trim());
}

function fieldsFor(stepId, vals) {
  return TOOL.fields(stepId, vals || {}) || [];
}
function stepMetaText(stepId) {
  const m = TOOL.stepMeta && TOOL.stepMeta[stepId];
  if (!m) return "";
  return typeof m === "function" ? m(values) : m;
}
function verdictOf(v) {
  if (TOOL.verdict) { const r = TOOL.verdict(v); if (r) return r; }
  return { t: "未完成", cls: "vb-neutral" };
}

function render() {
  if (view === "home") renderHome();
  else if (view === "wizard") renderWizard();
  else if (view === "report") renderReport();
  growTextareas(app);
  if (resetScrollNext) window.scrollTo(0, 0);
  resetScrollNext = false;
}
function goRender() { resetScrollNext = true; render(); }

function growTextareas(root) {
  root.querySelectorAll("textarea").forEach(growEl);
}
function growEl(el) {
  el.style.height = "auto";
  el.style.height = (el.scrollHeight + 2) + "px";
}

/* ---------- 首页 ---------- */
function renderHome() {
  const draft = loadDraft();
  const hasDraft = draft && draft[TOOL.titleKey];
  let reportsHtml = "";
  if (reports.length === 0) {
    reportsHtml = '<p class="empty-hint">还没有已保存的' + esc(TOOL.noun) + '。完成' + STEPS.length + '步流程后，报告会出现在这里。</p>';
  } else {
    reportsHtml = reports.map((r, i) => {
      const vd = verdictOf(r.values);
      return '<div class="report-item">' +
        '<div class="ri-main" data-open="' + i + '">' +
          '<div class="ri-title">' + esc(r.values[TOOL.titleKey] || "未命名") + '</div>' +
          '<div class="ri-meta">' + esc(r.date) + ' · ' + esc(vd.t) + '</div>' +
        '</div>' +
        '<button class="btn danger-ghost" data-del="' + i + '">删除</button>' +
      '</div>';
    }).join("");
  }

  app.innerHTML =
    '<div class="narrow">' +
    '<div class="hero">' +
      '<div class="kicker">' + esc(TOOL.kicker) + '</div>' +
      '<h1>' + esc(TOOL.h1) + '</h1>' +
      TOOL.intro.map(p => '<p>' + esc(p) + '</p>').join("") +
      '<p>这个工具把它拆成' + STEPS.length + '个小步骤：</p>' +
      '<ol>' +
        STEPS.map(st => '<li><strong>' + esc(st.title) + '</strong>——' + esc(st.desc) + '</li>').join("") +
      '</ol>' +
      '<div style="display:flex; gap:10px; flex-wrap:wrap; margin-top:14px;">' +
        '<button class="btn" id="btn-start">' + (hasDraft ? "继续上次的" + esc(TOOL.noun) + "（" + esc(draft[TOOL.titleKey]) + "）" : TOOL.startLabel) + '</button>' +
        (hasDraft ? '<button class="btn secondary" id="btn-fresh">重新开始</button>' : "") +
      '</div>' +
    '</div>' +

    '<div class="report-list">' +
      '<h2>已保存的' + esc(TOOL.noun) + '报告</h2>' + reportsHtml +
    '</div>' +
    '</div>';

  document.getElementById("btn-start").onclick = () => {
    values = loadDraft() || {};
    reviewEdit = -1; reviewIdx = null;
    view = "wizard"; cur = 0; goRender();
  };
  const fresh = document.getElementById("btn-fresh");
  if (fresh) fresh.onclick = () => {
    values = {}; saveDraft();
    reviewEdit = -1; reviewIdx = null;
    view = "wizard"; cur = 0; goRender();
  };
  app.querySelectorAll("[data-open]").forEach(el => {
    el.onclick = () => { viewingReport = reports[Number(el.dataset.open)]; view = "report"; goRender(); };
  });
  app.querySelectorAll("[data-del]").forEach(el => {
    el.onclick = (e) => {
      e.stopPropagation();
      if (!confirm("确定删除这份报告吗？此操作不可撤销。")) return;
      reports.splice(Number(el.dataset.del), 1);
      persistReports(); render();
    };
  });
}

/* ---------- 向导 ---------- */
function renderWizard() {
  const chips = STEPS.map((s, i) => {
    const cls = i === cur ? "active" : (i < cur ? "done" : "");
    const numTxt = i < cur ? "✓" : String(i + 1);
    let chip = '<span class="step-chip ' + cls + '" data-goto="' + i + '">' +
      '<span class="num">' + numTxt + '</span>' + esc(s.title) + '</span>';
    if (i < STEPS.length - 1) chip += '<span class="step-sep">›</span>';
    return chip;
  }).join("");

  const isLast = cur === STEPS.length - 1;
  const titleOk = !(STEPS[cur].id === TOOL.titleStep && !values[TOOL.titleKey]);
  const navRow =
    '<div class="nav-row">' +
      (cur > 0 ? '<button class="btn secondary" id="btn-prev">← 上一步</button>' : '<button class="btn secondary" id="btn-home">← 返回首页</button>') +
      '<span class="nav-note">' + (titleOk ? "" : esc(TOOL.titleNote)) + '</span>' +
      '<button class="btn" id="btn-next" ' + (titleOk ? "" : "disabled") + '>' + (isLast ? "保存并生成报告" : "下一步 →") + '</button>' +
    '</div>';

  const step = STEPS[cur];
  const meta = stepMetaText(step.id);
  const metaHtml = meta ? '<div class="meta-note">' + esc(meta) + '</div>' : "";
  const leftCard =
    '<div class="step-card">' +
      '<div class="step-head">' +
        '<span class="tag">第 ' + (cur + 1) + ' 步 · ' + esc(step.tag) + '</span>' +
        '<h2>' + esc(step.title) + '</h2>' +
      '</div>' +
      '<div class="step-body">' + metaHtml + stepBody(cur, false) + navRow + '</div>' +
    '</div>';

  let rightCol = "";
  if (cur > 0) {
    if (reviewIdx == null || reviewIdx < 0 || reviewIdx > cur - 1) reviewIdx = cur - 1;
    const i = reviewIdx;
    const editing = reviewEdit === i;
    const rMeta = stepMetaText(STEPS[i].id);
    const card =
      '<div class="review-card">' +
        '<div class="review-head">' +
          '<div class="rv-nav">' +
            (cur > 1 ? '<button class="rv-arrow" data-rvprev' + (i <= 0 ? " disabled" : "") + '>‹</button>' : "") +
            '<h3>回顾 · 第 ' + (i + 1) + ' 步 ' + esc(STEPS[i].title) + '（' + (i + 1) + '/' + cur + '）</h3>' +
            (cur > 1 ? '<button class="rv-arrow" data-rvnext' + (i >= cur - 1 ? " disabled" : "") + '>›</button>' : "") +
          '</div>' +
          '<button class="btn-edit ' + (editing ? "on" : "") + '" data-review="' + i + '">' + (editing ? "完成" : "编辑") + '</button>' +
        '</div>' +
        '<div class="review-body' + (editing ? "" : " locked") + '">' + (rMeta ? '<div class="meta-note">' + esc(rMeta) + '</div>' : "") + stepBody(i, !editing) + '</div>' +
      '</div>';
    rightCol = '<div class="review-col">' + card + '</div>';
  }

  app.innerHTML =
    '<div class="stepper">' + chips + '</div>' +
    (cur > 0 ? '<div class="split">' + leftCard + rightCol + '</div>' : leftCard);

  bindFieldEvents(app);
  app.querySelectorAll("[data-goto]").forEach(el => {
    el.onclick = () => { cur = Number(el.dataset.goto); reviewEdit = -1; reviewIdx = null; goRender(); };
  });
  app.querySelectorAll("[data-review]").forEach(el => {
    el.onclick = () => { reviewEdit = (reviewEdit === reviewIdx) ? -1 : reviewIdx; render(); };
  });
  app.querySelectorAll("[data-rvprev]").forEach(el => {
    el.onclick = () => { reviewIdx = Math.max(0, reviewIdx - 1); reviewEdit = -1; render(); };
  });
  app.querySelectorAll("[data-rvnext]").forEach(el => {
    el.onclick = () => { reviewIdx = Math.min(cur - 1, reviewIdx + 1); reviewEdit = -1; render(); };
  });
  const prev = document.getElementById("btn-prev");
  if (prev) prev.onclick = () => { cur--; reviewEdit = -1; reviewIdx = null; goRender(); };
  const home = document.getElementById("btn-home");
  if (home) home.onclick = () => { view = "home"; goRender(); };
  document.getElementById("btn-next").onclick = () => {
    if (isLast) finishReport();
    else { cur++; reviewEdit = -1; reviewIdx = null; goRender(); }
  };
}

/* ---------- 步骤主体 ---------- */
function stepBody(i, locked) {
  return renderFields(fieldsFor(STEPS[i].id, values), locked);
}

/* ---------- 字段渲染 ---------- */
function renderFields(fields, locked) {
  return fields.map(f => renderField(f, locked)).join("");
}

function renderField(f, locked) {
  if (f.showIf && !f.showIf(values)) return "";
  const lk = locked ? " locked" : "";
  const dis = locked ? " disabled" : "";
  const v = values[f.key];
  let body = "";
  if (f.type === "text") {
    body = '<input type="text" data-k="' + f.key + '"' + dis + ' value="' + esc(v || "") + '" placeholder="' + esc(f.placeholder || "") + '">';
  } else if (f.type === "textarea") {
    body = '<textarea data-k="' + f.key + '" rows="3"' + dis + ' placeholder="' + esc(f.placeholder || "") + '">' + esc(v || "") + '</textarea>';
  } else if (f.type === "radio") {
    body = '<div class="radio-group' + lk + '">' + f.options.map(op =>
      '<label class="radio-option ' + (v === op[0] ? "checked" : "") + lk + '">' +
        '<input type="radio" name="' + f.key + '" value="' + op[0] + '" ' + (v === op[0] ? "checked" : "") + dis + '>' +
        '<span>' + esc(op[1]) + (op[2] ? '<span class="opt-desc">' + esc(op[2]) + '</span>' : "") + '</span>' +
      '</label>').join("") + '</div>';
  } else if (f.type === "score") {
    body = '<div class="score-row' + lk + '">' +
      '<span class="score-endlabel">低</span>' +
      [0,1,2,3,4,5].map(n =>
        '<button type="button" class="score-btn ' + (Number(v) === n ? "checked" : "") + '" data-score="' + f.key + '" data-val="' + n + '"' + dis + '>' + n + '</button>'
      ).join("") +
      '<span class="score-endlabel right">高</span>' +
    '</div>';
  } else if (f.type === "list") {
    const rows = Array.isArray(v) ? v : [];
    const rowsHtml = rows.length === 0
      ? (f.emptyHint ? '<div class="list-empty-hint">' + esc(f.emptyHint) + '</div>' : "")
      : '<div class="list-rows">' + rows.map((row, ri) =>
          '<div class="list-row">' +
            '<span class="idx">' + (ri + 1) + '.</span>' +
            '<textarea data-lk="' + f.key + '" data-ri="' + ri + '" data-rk="t" rows="1"' + dis + ' placeholder="' + esc(f.itemPlaceholder || "") + '">' + esc(row.t || "") + '</textarea>' +
            '<button type="button" class="row-remove" data-rmrow="' + f.key + '" data-ri="' + ri + '" title="删除本条"' + dis + '>×</button>' +
          '</div>').join("") + '</div>';
    body = rowsHtml +
      '<button type="button" class="list-add" data-addrow="' + f.key + '"' + dis + '>＋ 添加一条</button>';
  } else if (f.type === "chipslist") {
    const rows = Array.isArray(v) ? v : [];
    const rowsHtml = rows.length === 0
      ? (f.emptyHint ? '<div class="list-empty-hint">' + esc(f.emptyHint) + '</div>' : "")
      : '<div class="list-rows">' + rows.map((row, ri) =>
          '<div class="list-row">' +
            '<span class="idx">' + (ri + 1) + '.</span>' +
            '<div class="grow" style="flex:1;display:flex;flex-direction:column;gap:8px;">' +
              '<textarea data-lk="' + f.key + '" data-ri="' + ri + '" data-rk="t" rows="1"' + dis + ' placeholder="' + esc(f.itemPlaceholder || "") + '">' + esc(row.t || "") + '</textarea>' +
              '<div class="row-cmp">' + f.chips.map(ch =>
                '<span class="chip ' + (row.c === ch[0] ? "checked" : "") + '" data-chip-lk="' + f.key + '" data-ri="' + ri + '" data-chip="' + ch[0] + '"' + dis + '>' + esc(ch[1]) + '</span>'
              ).join("") + '</div>' +
            '</div>' +
            '<button type="button" class="row-remove" data-rmrow="' + f.key + '" data-ri="' + ri + '" title="删除本条"' + dis + '>×</button>' +
          '</div>').join("") + '</div>';
    body = rowsHtml +
      '<button type="button" class="list-add" data-addrow="' + f.key + '"' + dis + '>＋ 添加一条</button>';
  } else if (f.type === "assess") {
    const rows = f.sources.reduce((acc, k) => acc.concat((Array.isArray(values[k]) ? values[k] : []).filter(x => (x.t || "").trim())), []);
    if (rows.length === 0) {
      body = '<div class="field-hint">' + esc(f.emptyHint || "请先在前面填写对应的条目，这里会自动生成逐条评定。") + '</div>';
    } else {
      const ans = v || {};
      body = rows.map((row, ri) => {
        const a = ans["i" + ri];
        return '<div class="as-row' + lk + '">' +
          '<div class="as-item">' + (ri + 1) + '. ' + esc(row.t || "") + '</div>' +
          '<div class="as-btns">' +
            f.options.map(op => {
              const pickedCls = a === op[0] ? " " + op[2] : "";
              return '<span class="as-btn' + pickedCls + '" data-assess="' + f.key + '" data-i="' + ri + '" data-v="' + op[0] + '"' + dis + '>' + esc(op[1]) + '</span>';
            }).join("") +
          '</div>' +
        '</div>';
      }).join("");
    }
  } else if (f.type === "note") {
    return '<div class="field' + lk + '"><label class="field-label">' + esc(f.label) + '</label><div class="meta-note">' + esc(f.text || "") + '</div></div>';
  }
  const hint = f.hint ? '<div class="field-hint">' + esc(f.hint) + '</div>' : "";
  return '<div class="field' + lk + '"><label class="field-label">' + esc(f.label) + '</label>' + body + hint + '</div>';
}

/* ---------- 字段事件 ---------- */
function bindFieldEvents(root) {
  root.querySelectorAll("input[data-k], textarea[data-k]").forEach(el => {
    el.addEventListener("input", () => {
      values[el.dataset.k] = el.value; saveDraft();
      growEl(el);
      if (el.dataset.k === TOOL.titleKey) {
        const btn = document.getElementById("btn-next");
        const note = document.querySelector(".nav-note");
        const ok = !!el.value.trim();
        if (btn) btn.disabled = !ok;
        if (note) note.textContent = ok ? "" : esc(TOOL.titleNote);
      }
    });
  });

  root.querySelectorAll('input[type="radio"]').forEach(el => {
    el.addEventListener("change", () => { values[el.name] = el.value; saveDraft(); render(); });
  });

  root.querySelectorAll(".score-btn").forEach(el => {
    el.onclick = () => {
      const k = el.dataset.score;
      values[k] = (values[k] === el.dataset.val) ? "" : el.dataset.val;
      saveDraft(); render();
    };
  });

  root.querySelectorAll(".as-btn").forEach(el => {
    el.onclick = () => {
      const k = el.dataset.assess;
      if (!values[k]) values[k] = {};
      const cur0 = values[k]["i" + el.dataset.i];
      values[k]["i" + el.dataset.i] = (cur0 === el.dataset.v) ? "" : el.dataset.v;
      saveDraft(); render();
    };
  });

  root.querySelectorAll("[data-addrow]").forEach(el => {
    el.onclick = () => {
      const k = el.dataset.addrow;
      if (!Array.isArray(values[k])) values[k] = [];
      values[k].push({}); saveDraft(); render();
    };
  });
  root.querySelectorAll("[data-rmrow]").forEach(el => {
    el.onclick = () => {
      values[el.dataset.rmrow].splice(Number(el.dataset.ri), 1); saveDraft(); render();
    };
  });
  root.querySelectorAll("textarea[data-lk]").forEach(el => {
    el.addEventListener("input", () => {
      values[el.dataset.lk][Number(el.dataset.ri)][el.dataset.rk] = el.value; saveDraft();
      growEl(el);
    });
  });
  root.querySelectorAll(".chip[data-chip-lk]").forEach(el => {
    el.onclick = () => {
      const row = values[el.dataset.chipLk][Number(el.dataset.ri)];
      row.c = (row.c === el.dataset.chip) ? "" : el.dataset.chip;
      saveDraft(); render();
    };
  });
}

/* ---------- 保存报告 ---------- */
function finishReport() {
  const clean = JSON.parse(JSON.stringify(values));
  Object.keys(clean).forEach(k => {
    const arr = clean[k];
    if (Array.isArray(arr) && arr.length && arr.every(x => typeof x === "object" && x !== null)) {
      clean[k] = arr.filter(row => Object.values(row).some(val => val != null && val !== ""));
    }
  });
  const report = { id: Date.now(), date: new Date().toLocaleString("zh-CN"), values: clean };
  reports.unshift(report);
  persistReports();
  localStorage.removeItem(LS_DRAFT);
  values = {};
  viewingReport = report;
  view = "report";
  goRender();
}

/* ---------- 报告 ---------- */
function rv(label, val) {
  const has = val != null && val !== "" && !(Array.isArray(val) && val.length === 0);
  const body = !has
    ? '<div class="rv-value empty">（未填写）</div>'
    : Array.isArray(val)
      ? '<ul class="rv-list">' + val.map(x => rvItemText(x)).join("") + '</ul>'
      : '<div class="rv-value">' + esc(val) + '</div>';
  return '<div class="rv-item"><div class="rv-label">' + esc(label) + '</div>' + body + '</div>';
}
function rvItemText(x) {
  if (typeof x === "object") {
    const chipMap = f_chipMap || {};
    const chipLabel = x.c && chipMap[x.c] ? "（" + chipMap[x.c] + "）" : "";
    return esc((x.t || "") + chipLabel);
  }
  return esc(x);
}
let f_chipMap = {};

function rvHtml(label, html) {
  const body = (html == null || html === "") ? '<div class="rv-value empty">（未填写）</div>' : html;
  return '<div class="rv-item"><div class="rv-label">' + esc(label) + '</div>' + body + '</div>';
}

function renderRV(f, v) {
  const val = v[f.key];
  if (f.type === "radio") {
    const opt = (f.options || []).find(o => o[0] === val);
    return rv(f.label, opt ? opt[1] : "");
  }
  if (f.type === "score") {
    return rv(f.label, (val != null && val !== "") ? val + " / 5" : "");
  }
  if (f.type === "list") {
    return rv(f.label, val);
  }
  if (f.type === "chipslist") {
    f_chipMap = {};
    (f.chips || []).forEach(ch => { f_chipMap[ch[0]] = ch[1]; });
    return rv(f.label, val);
  }
  if (f.type === "assess") {
    const rows = (f.sources || []).reduce((acc, k) => acc.concat((Array.isArray(v[k]) ? v[k] : []).filter(x => (x.t || "").trim())), []);
    const ans = val || {};
    const html = rows.map((row, ri) => {
      const a = ans["i" + ri];
      if (!a) return "";
      const opt = (f.options || []).find(o => o[0] === a);
      return '<div class="rv-as"><b>' + esc(opt ? opt[1] : "") + '</b> — ' + esc(row.t || "") + '</div>';
    }).join("");
    if (html) return rvHtml(f.label, html);
    return "";
  }
  if (f.type === "note") return "";
  return rv(f.label, val);
}

  }).join("");

  app.innerHTML =

    const text = reportToText(r);
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(text).then(() => flashCopied(), () => fallbackCopy(text));
    } else fallbackCopy(text);
  };
  function backHome() { view = "home"; viewingReport = null; goRender(); }
}

function reportToText(r) {
  const v = r.values;
  const vd = verdictOf(v);
  const L = [];
  L.push(TOOL.reportKicker.split(" · ")[0] + "：" + (v[TOOL.titleKey] || "未命名"));
  if (vd.t) L.push("评估结论：" + vd.t);
  if (v[TOOL.conclusionKey]) L.push("一句话结论：" + v[TOOL.conclusionKey]);
  L.push("");
  STEPS.forEach((st, si) => {
    const pairs = [];
    fieldsFor(st.id, v).forEach(f => {
      if (f.type === "note") return;
      const val = v[f.key];
      if (f.type === "radio") {
        const opt = (f.options || []).find(o => o[0] === val);
        if (opt) pairs.push([f.label, opt[1]]);
      } else if (f.type === "score") {
        if (val != null && val !== "") pairs.push([f.label, val + " / 5"]);
      } else if (f.type === "assess") {
        const rows = (f.sources || []).reduce((acc, k) => acc.concat((Array.isArray(v[k]) ? v[k] : []).filter(x => (x.t || "").trim())), []);
        const ans = val || {};
        rows.forEach((row, ri) => {
          const a = ans["i" + ri];
          if (a) {
            const opt = (f.options || []).find(o => o[0] === a);
            pairs.push([f.label + "：" + (row.t || ""), opt ? opt[1] : ""]);
          }
        });
      } else if (f.type === "chipslist") {
        const chipMap = {};
        (f.chips || []).forEach(ch => { chipMap[ch[0]] = ch[1]; });
        const items = (Array.isArray(val) ? val : []).map(x => (x.t || "") + (x.c && chipMap[x.c] ? "（" + chipMap[x.c] + "）" : ""));
        if (items.length) pairs.push([f.label, items]);
      } else {
        if (val != null && val !== "" && (!Array.isArray(val) || val.length)) pairs.push([f.label, val]);
      }
    });
    if (pairs.length) section_text(L, st.title, pairs);
  });
  return L.join("\n");

  function section_text(L2, name, pairs) {
    L2.push("【" + name + "】");
    pairs.forEach(([label, val]) => {
      if (val == null || val === "" || (Array.isArray(val) && val.length === 0)) return;
      if (Array.isArray(val)) {
        L2.push(label + "：");
        val.forEach(x => L2.push("  - " + (typeof x === "object" ? (x.t || "") : x)));
      } else L2.push(label + "：" + val);
    });
    L2.push("");
  }
}

function fallbackCopy(text) {
  const ta = document.createElement("textarea");
  ta.value = text;
  document.body.appendChild(ta);
  ta.select();
  try { document.execCommand("copy"); flashCopied(); } catch (e) {}
  document.body.removeChild(ta);
}
function flashCopied() {
  const btn = document.getElementById("btn-copy");
  if (!btn) return;
  const old = btn.textContent;
  btn.textContent = "已复制 ✓";
  setTimeout(() => { btn.textContent = old; }, 1500);
}
